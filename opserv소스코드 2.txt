#include "netprog.h"
#define MAXLINE 127

int main(int argc, char *argv[]){
    struct sockaddr_in servaddr,cliaddr;
    int listen_sock,accp_sock, addrlen,nbyte;
    int32_t num1,num2,num_1,num_2,result,result_;
    char buf[MAXLINE+1],op;

    if(argc != 2){
        printf("usage: %s port\n",argv[0]);
        exit(0);
    }

    if( (listen_sock = socket(PF_INET, SOCK_STREAM,0))<0){
        perror("socket fial");
        exit(0);
    }

    bzero( (char*)&servaddr, sizeof(servaddr));
    servaddr.sin_family = AF_INET;
    servaddr.sin_addr.s_addr = htonl(INADDR_ANY);
    servaddr.sin_port = htons(atoi(argv[1]));

    if(bind (listen_sock,(struct sockaddr*)&servaddr, sizeof(servaddr))<0){
        perror("bind fail");
        exit(0);
    }
    listen(listen_sock,5);

    puts("Server waiting...");

    while(1){
        accp_sock = accept(listen_sock,(struct sockaddr*)&cliaddr,&addrlen);
        if(accp_sock<0){
            perror("accept fail");
            exit(0);
        }
        puts("client connected!");


        read(accp_sock,buf,sizeof(buf));

        op =buf[0];
        memcpy(&num_1,buf+1,sizeof(num_1));
        memcpy(&num_2,buf+1+sizeof(num_2),sizeof(num_2));

        num1=ntohl(num_1);
        num2=ntohl(num_2);

        if(op =='+')
            result=num1+num2;
        else if(op == '-')
            result=num1-num2;

        result_ = htonl(result);
        memcpy(buf,&result_,sizeof(result_));

        write(accp_sock,buf,sizeof(result_));
        }
    close(accp_sock);

    close(listen_sock);
    return 0;
}
