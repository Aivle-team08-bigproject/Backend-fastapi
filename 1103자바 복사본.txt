상속하는방법:
class 자식_클래스 이름 extends 부모_클래스_이름{

}

overloading 다형성. 이름은 같으나 매개변수나 매개변수의 자료형 이런게 다름.

메소드 오버라이딩 덧붙여 쓴다. 부모클래스에서 멤버 변수나 멤버 함수는 물려받는데 
그 받은 것 중에서 자식 클래스에서 다시 정의해서 씀, 함수의 내용만 달라지지 매개변수의 개수 반환형, 이름은 같아햐함.

//Tv.java
public class Tv {
	protected	int channel; 
	protected	int volume;	
	protected	boolean onoff; 
	
	//생성자 #1 - 매개변수가 없는 생성자
	Tv(){ 
		channel = 10;
		volume = 20;
		onoff = false;
	}
	
	//생성자 #2 - 매개변수가 있는 생성자
	Tv(int pChannel,int pVolume, boolean pOnOff){
		channel = pChannel;
		volume = pVolume;
		onoff = pOnOff;
	}
	
	
	int GetChannel(){  //멤버함수
		return channel;
	}
	
	void SetChannel(int pChannel){// 멤버함수
		if(pChannel >=0 && pChannel <=50)
			{
				channel=pChannel;
				System.out.println("Channel has been changed to " + channel);
			}
		else
			System.out.println("Channel can not be changed to " + pChannel +" because of out of range [0..50]");
	}
	
	int GetVolume() { //멤버 함수
		return volume;
	}
	
	void SetVolume(int pVolume){ // 멤버함수
		if(pVolume >=0 && pVolume <=100)
			{
				volume=pVolume;
				System.out.println("Volume has been changed to " + volume);
			}
		else
			System.out.println("Volume can not be changed to " + pVolume +" because of out of range [0..100]");
	}
	
	boolean GetOnOff(){ //멤버 함수
		return onoff;
	}
	
	void SetOnOff(boolean pOnOff) { //멤버 함수
		onoff=pOnOff;
	}
	
	//@오버라이딩 된다
	void PrintClassName()
	{
		System.out.println("Class Name = Tv");
	}
	
}

//OLED.java
public class OLED extends Tv{
	private int USB;   			//USB 포트 갯수	[0..5]
	private int HDMI;  			//HDMI 포트 갯수	[0..10]
	private boolean XBOX; 		//XBOX 게임기 단자 설치 여부
	
	// 생성자 #1 - 매개변수가 없는 생성자
	public OLED() {
		super();	//부모 TV 클래스의 TV() 생성자를 호출함.
		
		USB=2;
		HDMI=3;
		XBOX=true;
	}
	
	//생성자 #2 - 매개변수가 있는 생성자
	public OLED(int pChannel,int pVolume,boolean pOnOff,int pUSB,int pHDMI,boolean pXBOX) 
	{
		super(pChannel,pVolume,pOnOff); //부모 TV 클래스의 TV(int pChannel,int pVolume,boolean pOnOff) 생성자를 호출함.
		
		USB=pUSB;
		HDMI=pHDMI;
		XBOX=pXBOX;
		
	}
	
	
	
	int GetUSB(){  //멤버함수
		return USB;
	}
	
	void SetUSB(int pUSB){// 멤버함수
			if((pUSB)>=0 && (pUSB<=5))
				{
					USB=pUSB;
					System.out.println("USB has been changed to"+USB);
				}
			else
				System.out.println("USB can not been changed to"+pUSB);
			
	}
	
	int GetHDMI(){  //멤버함수
		return HDMI;
	}
	
	void SetHDMI(int pHDMI){// 멤버함수
			if((pHDMI)>=0 && (pHDMI<=10))
				{
					HDMI=pHDMI;
					System.out.println("HDMI has been changed to"+HDMI);
				}
			else
				System.out.println("HDMI can not be changed to"+pHDMI);
			
	}
	
	boolean GetXBOX(){ //멤버 함수
		return XBOX;
	}
	
	void SetXBOX(boolean pXBOX) { //멤버 함수
		XBOX=pXBOX;
	}
	
	//@ 오버라이딩 된다.
	void PrintClassName()
	{
		System.out.println("Class Name = OLED");
	}
}

		for(i =0; i<=4; i++)
		{
			System.out.print(y[i]+", ");
		}
		
		System.out.println();
	
		for(i =0; i<=4; i++)
		{
			System.out.print(z[i]+", ");
		}
		
		System.out.println();
		
		MyArray.Swap(z, 1, 3); // y배열의 1번째 원소랑 3번째 원소 swap
		
		
		for(i =0; i<=4; i++)
		{
			System.out.print(z[i]+", ");
		}
	}

}

//MyArray.java


public class MyArray {
	public static <T> void Swap (T[] x,int i,int j)
	{
		T temp;
		
		temp = x[i];
		x[i]=x[j];
		x[j]=temp;
	}
	
	
}

//Main.java
import java.util.*;
import java.util.Vector;

public class Main {

	public static void main(String[] args) {
		
		
		Scanner input = new Scanner (System.in);
		
		Vector v = new Vector();
		int i;
		
		v.add(10); // add는 순번대로 들어오는대로 공간이생기고 값이 들어감. Vector 클래스는 정수형,실수형,문자형 상관없이 그냥 들어감.
		v.add(20.2);
		v.add("Hello");
	
		System.out.println("Vector v size = " + v.size()); //v.size 라는것이 Vector v에 지금 10,20, Hello 를 넣음으로써 3개의 공간이 생기는것.
		
		for( i=0; i < v.size(); i++)
		{
			System.out.println(v.get(i));
		}
	
		System.out.println();
		
		v.add(1, "Good Afternoon"); // add(순번, 집어넣을것) 해주면 1번째(배열처럼 생각) 자리에 들어가짐.
	
		for( i=0; i < v.size(); i++)
		{
			System.out.println(v.get(i));
		}
		
		System.out.println();

	}
}
